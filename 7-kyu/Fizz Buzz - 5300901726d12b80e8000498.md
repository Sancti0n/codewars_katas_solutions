## Python
```py
def fizzbuzz(n):
    t = []
    for i in range(1, n+1):
        if i%15 == 0: t.append('FizzBuzz')
        elif i%5 == 0: t.append('Buzz')
        elif i%3 == 0: t.append('Fizz')
        else: t.append(i)
    return t
```

## JavaScript
```js
function fizzbuzz(n) {
  let t = [];
  for (let i=1;i<=n;i++) {
    if (i%15 === 0) t.push('FizzBuzz');
    else if (i%5 == 0) t.push('Buzz');
    else if (i%3 == 0) t.push('Fizz');
    else t.push(i)
  }
  return t
}
```
